DATABASE INFORMATION
Database name: reactt_user_management

Tables observed in MySQL Workbench:
1. users
   The backend code expects these columns: id, name, email, company.
2. auth_users
   Columns shown in Workbench: id (auto-increment primary key), email (unique),
   password_hash, created_at. This is the table used by server.js for registration
   and login.
3. auth_sessions
   Columns shown in Workbench: id (auto-increment primary key), auth_user_id,
   token (unique), expires_at, created_at; auth_user_id is a foreign key to
   auth_users(id). The current JWT code does not query this table.

IMPORTANT DATABASE NOTE
backend/database.sql is included below because it exists in the project, but it
creates an older app_users table. Do NOT run it for the current setup, because
server.js currently uses auth_users.

Frontend development tools: Vite, ESLint, @vitejs/plugin-react, and React ESLint packages.
Backend (backend/package.json): express, mysql2, cors, bcryptjs, dotenv, jsonwebtoken.

require('dotenv').config()

const express = require('express')
const mysql = require('mysql2')
const cors = require('cors')
const bcrypt = require('bcryptjs')
const jwt = require('jsonwebtoken')

const app = express()
const PORT = process.env.PORT || 5000
const JWT_SECRET = process.env.JWT_SECRET

if (!JWT_SECRET) {
  throw new Error('JWT_SECRET is missing. Add it to backend/.env before starting the server.')
}

app.use(cors())
app.use(express.json())

// A pool reuses database connections instead of opening one connection per request.
const db = mysql.createPool({
  host: process.env.DB_HOST || 'localhost',
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
})

const query = (sql, values = []) =>
  new Promise((resolve, reject) => {
    db.query(sql, values, (error, results) => {
      if (error) return reject(error)
      resolve(results)
    })
  })

// Checks the token sent by React in: Authorization: Bearer <token>
function authenticateToken(req, res, next) {
  const authorization = req.headers.authorization
  const token = authorization && authorization.split(' ')[1]

  if (!token) return res.status(401).json({ error: 'Login required' })

  jwt.verify(token, JWT_SECRET, (error, user) => {
    if (error) return res.status(403).json({ error: 'Your session has expired. Please log in again.' })
    req.user = user
    next()
  })
}

const makeUserResponse = (user) => ({ id: user.id, email: user.email })

// Creates a login account. Passwords are hashed; they are never stored as plain text.
app.post('/api/auth/register', async (req, res) => {
  const { email, password } = req.body

  if (!email || !password) {
    return res.status(400).json({ error: 'Email and password are required' })
  }
  if (password.length < 8) {
    return res.status(400).json({ error: 'Password must be at least 8 characters long' })
  }

  try {
    const existingUsers = await query('SELECT id FROM auth_users WHERE email = ?', [email])
    if (existingUsers.length) return res.status(409).json({ error: 'An account with this email already exists' })

    const passwordHash = await bcrypt.hash(password, 12)
    const result = await query(
      'INSERT INTO auth_users (email, password_hash) VALUES (?, ?)',
      [email, passwordHash]
    )
    const user = { id: result.insertId, email }
    const token = jwt.sign(user, JWT_SECRET, { expiresIn: '1h' })

    res.status(201).json({ token, user })
  } catch (error) {
    console.error('Registration failed:', error.message)
    res.status(500).json({ error: 'Could not create account' })
  }
})

// Verifies email + password and creates a signed JWT session token.
app.post('/api/auth/login', async (req, res) => {
  const { email, password } = req.body
  if (!email || !password) return res.status(400).json({ error: 'Email and password are required' })

  try {
    const users = await query('SELECT * FROM auth_users WHERE email = ?', [email])
    const user = users[0]
    if (!user || !(await bcrypt.compare(password, user.password_hash))) {
      return res.status(401).json({ error: 'Incorrect email or password' })
    }

    const safeUser = makeUserResponse(user)
    const token = jwt.sign(safeUser, JWT_SECRET, { expiresIn: '1h' })
    res.json({ token, user: safeUser })
  } catch (error) {
    console.error('Login failed:', error.message)
    res.status(500).json({ error: 'Could not log in' })
  }
})

// Every dashboard API below this line requires a valid JWT.
app.use('/api/users', authenticateToken)

app.get('/api/users', async (req, res) => {
  try {
    const results = await query('SELECT * FROM users')
    res.json(results.map((user) => ({ ...user, company: { name: user.company } })))
  } catch (error) {
    res.status(500).json({ error: 'Failed to fetch users' })
  }
})

app.post('/api/users', async (req, res) => {
  const { name, email, company } = req.body
  if (!name || !email || !company) return res.status(400).json({ error: 'Name, email, and company are required' })

  try {
    const result = await query('INSERT INTO users (name, email, company) VALUES (?, ?, ?)', [name, email, company])
    res.status(201).json({ id: result.insertId, name, email, company })
  } catch (error) {
    res.status(500).json({ error: 'Failed to create user' })
  }
})

app.put('/api/users/:id', async (req, res) => {
  const { name, email, company } = req.body
  if (!name || !email || !company) return res.status(400).json({ error: 'Name, email, and company are required' })

  try {
    const result = await query('UPDATE users SET name = ?, email = ?, company = ? WHERE id = ?', [name, email, company, req.params.id])
    if (!result.affectedRows) return res.status(404).json({ error: 'User not found' })
    res.json({ id: Number(req.params.id), name, email, company })
  } catch (error) {
    res.status(500).json({ error: 'Failed to update user' })
  }
})

app.delete('/api/users/:id', async (req, res) => {
  try {
    const result = await query('DELETE FROM users WHERE id = ?', [req.params.id])
    if (!result.affectedRows) return res.status(404).json({ error: 'User not found' })
    res.json({ message: 'User deleted successfully' })
  } catch (error) {
    res.status(500).json({ error: 'Failed to delete user' })
  }
})

app.listen(PORT, () => console.log(`Server is running on http://localhost:${PORT}`))

import { useEffect, useState } from 'react'
import { useNavigate } from 'react-router-dom'
import './App.css'
import { apiFetch } from './api'
import UserList from './components/UserList'
import SearchFilter from './components/SearchFilter'
import UserForm from './components/UserForm'
import Pagination from './components/Pagination'
function App() {
  const navigate = useNavigate()
  const [users, setUsers] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState('')
  const [search, setSearch] = useState('')
  const [companyFilter, setCompanyFilter] = useState('All')
  const [currentPage, setCurrentPage] = useState(1)

  const [formData, setFormData] = useState({
    name: '',
    email: '',
    company: '',
  })

  const [editingUserId, setEditingUserId] = useState(null)

  const usersPerPage = 3

  useEffect(() => {
  apiFetch('/api/users')
      .then((response) => {
        if (!response.ok) {
          throw new Error('Failed to fetch users')
        }

        return response.json()
      })
      .then((data) => {
        setUsers(data)
      })
      .catch((error) => {
        setError(error.message)
      })
      .finally(() => {
        setLoading(false)
      })
  }, [])

  const filteredUsers = users.filter((user) => {
    const searchText = search.toLowerCase()

    const matchesSearch =
      user.name.toLowerCase().includes(searchText) ||
      user.email.toLowerCase().includes(searchText)

    const matchesCompany =
      companyFilter === 'All' || user.company.name === companyFilter

    return matchesSearch && matchesCompany
  })

  const totalPages = Math.ceil(filteredUsers.length / usersPerPage)

  const startIndex = (currentPage - 1) * usersPerPage

  const currentUsers = filteredUsers.slice(
    startIndex,
    startIndex + usersPerPage
  )

  const companies = [
    'All',
    ...new Set(users.map((user) => user.company.name)),
  ]

  const handleSearchChange = (event) => {
    setSearch(event.target.value)
    setCurrentPage(1)
  }
  const handleFilterChange = (event) => {
    setCompanyFilter(event.target.value)
    setCurrentPage(1)
  }

  const handleInputChange = (event) => {
    const { name, value } = event.target

    setFormData({
      ...formData,
      [name]: value,
    })
  }

  const handleSubmit = async (event) => {
  event.preventDefault()

  if (!formData.name || !formData.email || !formData.company) {
    alert('Please fill all fields')
    return
  }

  try {
    if (editingUserId) {
      const response = await apiFetch(
        `/api/users/${editingUserId}`,
        {
          method: 'PUT',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            name: formData.name,
            email: formData.email,
            company: formData.company,
          }),
        }
      )

      if (!response.ok) {
        throw new Error('Failed to update user')
      }

      const updatedUser = await response.json()

      setUsers(
        users.map((user) =>
          user.id === editingUserId
            ? {
                ...user,
                ...updatedUser,
                company: {
                  name: updatedUser.company,
                },
              }
            : user
        )
      )

      setEditingUserId(null)
    } else {
      const response = await apiFetch(
        '/api/users',
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            name: formData.name,
            email: formData.email,
            company: formData.company,
          }),
        }
      )

      if (!response.ok) {
        throw new Error('Failed to create user')
      }

      const newUser = await response.json()

      const formattedUser = {
        ...newUser,
        company: {
          name: newUser.company,
        },
      }

      setUsers([...users, formattedUser])
    }

    setFormData({
      name: '',
      email: '',
      company: '',
    })

    setCurrentPage(1)
  } catch (error) {
    alert(error.message)
  }
}
  const handleEdit = (user) => {
    setEditingUserId(user.id)

    setFormData({
      name: user.name,
      email: user.email,
      company: user.company.name,
    })
  }

  const handleDelete = async (id) => {
    const confirmed = window.confirm(
      'Are you sure you want to delete this user?'
    )

    if (!confirmed) {
      return
    }

    try {
      const response = await apiFetch(
        `/api/users/${id}`,
        {
          method: 'DELETE',
        }
      )

      if (!response.ok) {
        throw new Error('Failed to delete user')
      }

      setUsers(users.filter((user) => user.id !== id))
    } catch (error) {
      alert(error.message)
    }
  }

  const handleCancelEdit = () => {
    setEditingUserId(null)

    setFormData({
      name: '',
      email: '',
      company: '',
    })
  }

  const handleLogout = () => {
    localStorage.removeItem('token')
    localStorage.removeItem('currentUser')
    navigate('/login')
  }

return (
  <  div className="app">
    <div className="header">
      <h1>User Management Dashboard</h1>
      <p>Manage your users securely with React, Express, MySQL, and JWT.</p>
      <button type="button" className="logout" onClick={handleLogout}>
        Log out
      </button>
    </div>
<UserForm
  formData={formData}
  editingUserId={editingUserId}
  onInputChange={handleInputChange}
  onSubmit={handleSubmit}
  onCancel={handleCancelEdit}
/>
    
     <SearchFilter
  search={search}
  companyFilter={companyFilter}
  companies={companies}
  onSearchChange={handleSearchChange}
  onFilterChange={handleFilterChange}
/>

   <p>
  Showing {currentUsers.length} of {filteredUsers.length} matching users
</p>

{error ? (
  <div className="empty">
    <p>{error}</p>
  </div>
 ) : loading ? (
  <div className="user-list">
    {[1, 2, 3].map((item) => (
      <div className="skeleton-card" key={item}>
        <div className="skeleton skeleton-title"></div>
        <div className="skeleton skeleton-text"></div>
        <div className="skeleton skeleton-text"></div>
        <div className="skeleton skeleton-button"></div>
      </div>
    ))}
  </div>
): currentUsers.length === 0 ? (
  <div className="empty">
    <p>No users found.</p>
  </div>
) : (
  <UserList
    users={currentUsers}
    onEdit={handleEdit}
    onDelete={handleDelete}
  />
)}
{totalPages > 1 && (
  <Pagination
    currentPage={currentPage}
    totalPages={totalPages}
    onPrevious={() => setCurrentPage(currentPage - 1)}
    onNext={() => setCurrentPage(currentPage + 1)}
  />
)}
  </div>
)
}

export default App

import { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import './Login.css'
import { apiFetch } from './api'

function Login() {
  const navigate = useNavigate()
  const [isRegistering, setIsRegistering] = useState(false)
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [error, setError] = useState('')
  const [isSubmitting, setIsSubmitting] = useState(false)

  const handleSubmit = async (event) => {
    event.preventDefault()
    setError('')
    setIsSubmitting(true)

    try {
      const response = await apiFetch(
        isRegistering ? '/api/auth/register' : '/api/auth/login',
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ email, password }),
        }
      )
      const data = await response.json()

      if (!response.ok) throw new Error(data.error || 'Something went wrong')

      // localStorage lets the session survive a browser refresh.
      localStorage.setItem('token', data.token)
      localStorage.setItem('currentUser', JSON.stringify(data.user))
      navigate('/dashboard')
    } catch (requestError) {
      setError(requestError.message)
    } finally {
      setIsSubmitting(false)
    }
  }

  const switchMode = () => {
    setIsRegistering(!isRegistering)
    setError('')
  }

  return (
    <div className="login-container">
      <div className="login-box">
        <h1>{isRegistering ? 'Create account' : 'Welcome '}</h1>
        <p>{isRegistering ? 'Create an account to access the dashboard.' : 'Log in to manage users.'}</p>

        <form onSubmit={handleSubmit}>
          <div className="form-group">
            <label htmlFor="email">Email</label>
            <input id="email" type="email" value={email} onChange={(event) => setEmail(event.target.value)} required />
          </div>

          <div className="form-group">
            <label htmlFor="password">Password</label>
            <input id="password" type="password" minLength="8" value={password} onChange={(event) => setPassword(event.target.value)} required />
          </div>

          {error && <p className="login-error">{error}</p>}
          <button type="submit" disabled={isSubmitting}>
            {isSubmitting ? 'Please waitâ€¦' : isRegistering ? 'Create account' : 'Log in'}
          </button>
        </form>

        <button type="button" className="mode-switch" onClick={switchMode}>
          {isRegistering ? 'Already have an account? Log in' : 'Need an account? Register'}
        </button>
      </div>
    </div>
  )
}

export default Login

const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000'

export async function apiFetch(path, options = {}) {
  const token = localStorage.getItem('token')
  const headers = new Headers(options.headers)

  if (token) headers.set('Authorization', `Bearer ${token}`)

  const response = await fetch(`${API_URL}${path}`, { ...options, headers })

  // An expired/invalid token must not leave the app looking logged in.
  if (response.status === 401 || response.status === 403) {
    localStorage.removeItem('token')
    localStorage.removeItem('currentUser')
  }

  return response
}
